const SHEET_NAME = '案例資料'; function getOrCreateSheet(id) { const ss = SpreadsheetApp.openById(id); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['id','keydate','saleDate','surgery','staff','branch','name','code', 'level','amount','status','card','story','path', 'portrait','photo','video','aki','cbPre','cbPost']); } return sh; } function doPost(e) { const data = JSON.parse(e.postData.contents); const sh = getOrCreateSheet(data.sheetId); let result = {}; if (data.action === 'getAll') { const rows = sh.getDataRange().getValues(); const headers = rows[0]; const cases = rows.slice(1).map(r => { const obj = {}; headers.forEach((h, i) => { let v = r[i]; if (v === 'TRUE' || v === true) v = true; else if (v === 'FALSE' || v === false) v = false; else if (typeof v === 'number' && ['amount'].includes(h)) v = Number(v); obj[h] = v; }); return obj; }).filter(c => c.id); result = { cases }; } else if (data.action === 'upsert') { const c = data.case; const rows = sh.getDataRange().getValues(); const headers = rows[0]; const idCol = headers.indexOf('id'); let found = -1; for (let i = 1; i < rows.length; i++) { if (String(rows[i][idCol]) === String(c.id)) { found = i; break; } } const row = headers.map(h => { const v = c[h]; return (v === true || v === false) ? String(v).toUpperCase() : (v ?? ''); }); if (found >= 0) sh.getRange(found + 1, 1, 1, row.length).setValues([row]); else sh.appendRow(row); result = { ok: true }; } else if (data.action === 'batchUpsert') { const cases = data.cases; const rows = sh.getDataRange().getValues(); const headers = rows[0]; cases.forEach(c => { const row = headers.map(h => { const v = c[h]; return (v === true || v === false) ? String(v).toUpperCase() : (v ?? ''); }); sh.appendRow(row); }); result = { ok: true }; } else if (data.action === 'update') { const rows = sh.getDataRange().getValues(); const headers = rows[0]; const idCol = headers.indexOf('id'); const fCol = headers.indexOf(data.field); if (fCol < 0) { result = { error: 'field not found' }; } else { for (let i = 1; i < rows.length; i++) { if (String(rows[i][idCol]) === String(data.id)) { let v = data.value; if (v === true || v === false) v = String(v).toUpperCase(); sh.getRange(i + 1, fCol + 1).setValue(v); break; } } result = { ok: true }; } } else if (data.action === 'delete') { const rows = sh.getDataRange().getValues(); const headers = rows[0]; const idCol = headers.indexOf('id'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idCol]) === String(data.id)) { sh.deleteRow(i + 1); break; } } result = { ok: true }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
